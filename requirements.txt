      <h1>OpsAgentAI</h1>
      <p>Local, self-hosted agent using Ollama.</p>

      <textarea id="prompt" placeholder="Ask the agent to inspect files, shell commands, or project structure..."></textarea>
      <button id="runButton">Run agent</button>
      <div class="status" id="status">Ready.</div>
      <div class="response" id="response">No output yet.</div>
    </div>

    <script>
      const promptInput = document.getElementById('prompt');
      const runButton = document.getElementById('runButton');
      const responseBox = document.getElementById('response');
      const statusBox = document.getElementById('status');

      runButton.addEventListener('click', async () => {
        const prompt = promptInput.value.trim();
        if (!prompt) {
          statusBox.textContent = 'Please enter a prompt.';
          return;
        }

        statusBox.textContent = 'Running agent...';
        responseBox.textContent = 'Working...';

        try {
          const res = await fetch('/api/chat', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ prompt })
          });

          const data = await res.json();

          if (!res.ok) {
            throw new Error(data.error || 'Request failed');
          }

          responseBox.textContent = data.reply || 'No result returned.';
          statusBox.textContent = 'Done.';
        } catch (err) {
          responseBox.textContent = 'Error: ' + err.message;
          statusBox.textContent = 'Failed.';
        }
      });
    </script>
